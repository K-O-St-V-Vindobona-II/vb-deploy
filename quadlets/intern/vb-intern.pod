# Quadlet defaults pods to ExitPolicy=stop, which exits the pod
# cleanly (exit code 0) once its last container disappears - podman
# auto-update relies on exactly that (stop+recreate one container),
# and a clean exit does not trigger the pod's own Restart=on-failure.
# continue keeps the pod (and its infra container) alive so the
# recreated container can rejoin it instead of hitting "Dependency
# failed". Set via PodmanArgs rather than the dedicated ExitPolicy=
# key: that key was only added to the Quadlet parser in Podman 5.6,
# while the underlying --exit-policy flag on "podman pod create" is
# much older and PodmanArgs has supported the [Pod] section since
# early Quadlet versions - this keeps the fix working on distributions
# whose packaged Podman predates the dedicated key.
PodmanArgs=--exit-policy=continue

[Install]
WantedBy=default.target
